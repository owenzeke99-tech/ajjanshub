-- Tên file lưu key cục bộ (Dùng để nhớ key trong 24h)
local KEY_FILE = "AjjansHub_SavedKey.json"

-- Hàm kiểm tra xem đã lưu key và còn hạn 24h hay không
local function hasValidSavedKey()
    if readfile and isfile and isfile(KEY_FILE) then
        local success, data = pcall(function()
            return game:GetService("HttpService"):JSONDecode(readfile(KEY_FILE))
        end)
        if success and data and data.key == CORRECT_KEY and data.time then
            -- 86400 giây = 24 giờ
            if os.time() - data.time < 86400 then
                return true
            end
        end
    end
    return false
end

-- Hàm lưu key khi nhập đúng
local function saveKey()
    if writefile then
        local data = {
            key = CORRECT_KEY,
            time = os.time()
        }
        pcall(function()
            writefile(KEY_FILE, game:GetService("HttpService"):JSONEncode(data))
        end)
    end
end

local dragging, dragInput, dragStart, startPos
MainFrame.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        dragging = true
        dragStart = input.Position
        startPos = MainFrame.Position
        input.Changed:Connect(function()
            if input.UserInputState == Enum.UserInputState.End then
                dragging = false
            end
        end)
    end
end)
MainFrame.InputChanged:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
        dragInput = input
    end
end)
UserInputService.InputChanged:Connect(function(input)
    if input == dragInput and dragging then
        local delta = input.Position - dragStart
        MainFrame.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
    end
end)

-- Logic kiểm tra Key kèm hiệu ứng lắc mượt mà và tự xóa chữ khi sai
CheckKeyBtn.MouseButton1Click:Connect(function()
    local enteredKey = TextBox.Text
    
    if enteredKey == CORRECT_KEY then
        StatusLabel.TextColor3 = Color3.fromRGB(50, 200, 100)
        StatusLabel.Text = "Key correct, loading..."
        
        -- Lưu key lại để dùng trong 24 giờ tới
        saveKey()
        
        task.wait(0.8)
        
        ScreenGui:Destroy()
        
        local success, err = pcall(function()
            loadstring(game:HttpGet(SCRIPT_URL))()
        end)
        
        if not success then
            warn("Error loading script: " .. tostring(err))
        end
    else
        StatusLabel.TextColor3 = Color3.fromRGB(235, 50, 50)
        StatusLabel.Text = "Invalid key! Please try again."
        
        -- Xóa sạch chữ trong ô nhập key khi sai
        TextBox.Text = ""
        
        -- Hiệu ứng lắc nhẹ mượt mà (Smooth Shake Animation)
        local originalPos = MainFrame.Position
        local tweenInfo = TweenInfo.new(0.04, Enum.EasingStyle.Sine, Enum.EasingDirection.InOut)
        
        for i = 1, 2 do
            TweenService:Create(MainFrame, tweenInfo, {Position = originalPos + UDim2.new(0, 8, 0, 0)}):Play()
            task.wait(0.04)
            TweenService:Create(MainFrame, tweenInfo, {Position = originalPos - UDim2.new(0, 8, 0, 0)}):Play()
            task.wait(0.04)
        end
        TweenService:Create(MainFrame, tweenInfo, {Position = originalPos}):Play()
    end
end)
